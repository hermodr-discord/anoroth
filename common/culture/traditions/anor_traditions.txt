tradition_imperial_soldiers = { #Vathîrin Oúrapivêlis (OLD NAME)
	category = regional

	layers = {
		0 = martial
		1 = mena
		4 = knight.dds
	}
	
	is_shown = {
		OR = {
			this = culture:anorothic
			any_parent_culture_or_above = {
				this = culture:anorothic
			}
		}
	}
	can_pick = {
		OR = {
			this = culture:anorothic
			any_parent_culture_or_above = {
				this = culture:anorothic
			}
		}
	}
	
	parameters = {
		unlock_maa_anoroth_sun_knights = yes
		unlock_maa_anoroth_legionnaires = yes
		martial_education_more_valued = yes
	}
	character_modifier = {
		heavy_cavalry_max_size_add = 2
		heavy_infantry_max_size_add = 2
		counter_efficiency = 0.1
		knight_limit = 2
		negate_prowess_penalty_add = 5
		accolade_glory_gain_mult = 0.2
		glory_hound_opinion = 20
	}
	
	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_bellicose = { is_in_list = traits }
						culture_pillar:ethos_courtly = { is_in_list = traits }
						culture_pillar:ethos_communal = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_bellicose_courtly_or_communal_desc
				}
			}
			if = {
				limit = {
					NOT = {
						scope:character = {
							any_knight = {
								count >= 6
								prowess >= 12
							}
						}
					}
				}
				add = {
					value = tradition_unfulfilled_criteria_penalty
					desc = knights_with_prowess_desc
				}
			}
			
			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = { value = 400 }
}
